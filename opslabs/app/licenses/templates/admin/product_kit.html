"""

Generated by {{ site_name }} for product {{ product.code }}
Drop this file into your project root.

import os, json, threading, time, socket, requests
from datetime import datetime, timedelta

# ── Config ────────────────────────────────────────────────────────
LICENSE_SERVER  = "{{ base_url }}"
PRODUCT_CODE    = "{{ product.code }}"
LICENSE_FILE    = os.path.join(os.path.dirname(__file__), "license.txt")
HEARTBEAT_EVERY = 3600    # seconds between heartbeats
VALIDATE_TIMEOUT= 8       # seconds before giving up on server
GRACE_PERIOD    = 86400   # 24h grace if server unreachable
CACHE_FILE      = os.path.join(os.path.dirname(__file__), "instance/.lc_{{ product.code|lower }}")

# ── State ─────────────────────────────────────────────────────────
_state = {
    "valid": False, "checked": False, "license_key": None,
    "activation_id": None, "customer": None, "tier": None,
    "features": [], "expires_at": None, "days_left": None,
    "error": None, "last_ok": None, "server_down": False,
}
_lock = threading.Lock()

def _read_key():
    if not os.path.exists(LICENSE_FILE): return None
    try:
        for line in open(LICENSE_FILE):
            line = line.strip()
            if line and not line.startswith("#"): return line.upper()
    except: pass
    return None

def _hardware_id():
    try:
        import uuid
        mac = ':'.join(['{:02x}'.format((uuid.getnode() >> i) & 0xff)
                        for i in range(0,48,8)][::-1])
        return f"{socket.gethostname()}-{mac}"
    except: return socket.gethostname()

def _save_cache(data):
    try:
        os.makedirs(os.path.dirname(CACHE_FILE), exist_ok=True)
        json.dump({**data, "_cached_at": datetime.utcnow().isoformat()}, open(CACHE_FILE,"w"))
    except: pass

def _load_cache():
    try:
        data = json.load(open(CACHE_FILE))
        if datetime.utcnow() - datetime.fromisoformat(data["_cached_at"]) < timedelta(seconds=GRACE_PERIOD):
            return data
    except: pass
    return None

def validate_license(key=None):
    global _state
    key = key or _read_key()
    if not key:
        with _lock: _state.update({"valid":False,"checked":True,"error":"No licence key found in license.txt"})
        return False
    payload = {"license_key": key, "hardware_id": _hardware_id(), "domain": socket.gethostname()}
    try:
        data = requests.post(f"{LICENSE_SERVER}/api/validate", json=payload, timeout=VALIDATE_TIMEOUT).json()
        if data.get("valid"):
            act_id = None
            try:
                ar = requests.post(f"{LICENSE_SERVER}/api/activate",
                                   json={**payload,"hostname":socket.gethostname()}, timeout=VALIDATE_TIMEOUT).json()
                act_id = ar.get("activation_id")
            except: pass
            with _lock:
                _state.update({"valid":True,"checked":True,"license_key":key,
                               "activation_id":act_id,"customer":data.get("customer"),
                               "tier":data.get("tier_name"),"features":data.get("features",[]),
                               "expires_at":data.get("expires_at"),"days_left":data.get("days_until_expiry"),
                               "error":None,"last_ok":datetime.utcnow(),"server_down":False})
            _save_cache(data)
            print(f"[{{ product.code }}] ✓ Valid — {data.get('customer','?')} · {data.get('tier_name','')} · features: {data.get('features',[])}")
            return True
        err = data.get("error","Licence invalid")
        with _lock: _state.update({"valid":False,"checked":True,"error":err})
        print(f"[{{ product.code }}] ✗ {err}")
        return False
    except requests.exceptions.ConnectionError:
        return _grace("Cannot reach licence server")
    except requests.exceptions.Timeout:
        return _grace("Licence server timed out")
    except Exception as e:
        return _grace(str(e))

def _grace(reason):
    cache = _load_cache()
    if cache and cache.get("valid"):
        h = round((timedelta(seconds=GRACE_PERIOD)-(datetime.utcnow()-datetime.fromisoformat(cache["_cached_at"]))).total_seconds()/3600,1)
        with _lock:
            _state.update({"valid":True,"checked":True,
                           "features":cache.get("features",[]),
                           "error":f"⚠ Server offline — grace period ({h}h left)","server_down":True})
        return True
    with _lock: _state.update({"valid":False,"checked":True,"error":f"Server unreachable: {reason}","server_down":True})
    return False

def send_heartbeat():
    with _lock: key, act_id = _state.get("license_key"), _state.get("activation_id")
    if not key: return
    try:
        data = requests.post(f"{LICENSE_SERVER}/api/heartbeat",
                             json={"license_key":key,"activation_id":act_id}, timeout=VALIDATE_TIMEOUT).json()
        if data.get("valid"):
            with _lock: _state["last_ok"] = datetime.utcnow(); _state["server_down"] = False
    except: pass

def _heartbeat_loop():
    time.sleep(HEARTBEAT_EVERY)
    while True:
        try: send_heartbeat()
        except: pass
        time.sleep(HEARTBEAT_EVERY)

def is_valid():
    with _lock: return _state["valid"]

def get_state():
    with _lock: return dict(_state)

def has_feature(code):
    """Check if a specific feature is enabled for this licence."""
    with _lock: return code in _state.get("features", [])

def init_license(app=None):
    """Call this on startup. Pass your Flask app to block if invalid."""
    ok = validate_license()
    t = threading.Thread(target=_heartbeat_loop, daemon=True, name="LicenceHeartbeat_{{ product.code }}")
    t.start()
    if app and not ok:
        @app.before_request
        def _block():
            from flask import render_template
            return render_template("license_invalid.html", error=get_state().get("error")), 403
    return ok

 * {{ product.name }} — Licence Check (Node.js)
 * Generated by {{ site_name }} for product {{ product.code }}
 * npm install axios
 */

const axios = require('axios');
const fs    = require('fs');
const os    = require('os');
const path  = require('path');

const LICENSE_SERVER  = '{{ base_url }}';
const PRODUCT_CODE    = '{{ product.code }}';
const LICENSE_FILE    = path.join(__dirname, 'license.txt');
const HEARTBEAT_MS    = 3600 * 1000;
const VALIDATE_TIMEOUT= 8000;

let _state = {
  valid: false, checked: false, licenseKey: null,
  activationId: null, customer: null, tier: null,
  features: [], expiresAt: null, error: null,
};

function readKey() {
  try {
    return fs.readFileSync(LICENSE_FILE, 'utf8')
      .split('\n').map(l => l.trim())
      .find(l => l && !l.startsWith('#'))?.toUpperCase() || null;
  } catch { return null; }
}

async function validateLicense(key) {
  key = key || readKey();
  if (!key) {
    _state = { ..._state, valid: false, checked: true, error: 'No licence key found in license.txt' };
    return false;
  }
  const payload = { license_key: key, hardware_id: os.hostname(), domain: os.hostname() };
  try {
    const { data } = await axios.post(`${LICENSE_SERVER}/api/validate`, payload, { timeout: VALIDATE_TIMEOUT });
    if (data.valid) {
      let activationId = null;
      try {
        const ar = await axios.post(`${LICENSE_SERVER}/api/activate`,
          { ...payload, hostname: os.hostname() }, { timeout: VALIDATE_TIMEOUT });
        activationId = ar.data.activation_id;
      } catch {}
      _state = { valid: true, checked: true, licenseKey: key, activationId,
                 customer: data.customer, tier: data.tier_name,
                 features: data.features || [], expiresAt: data.expires_at, error: null };
      console.log(`[{{ product.code }}] ✓ Valid — ${data.customer} · ${data.tier_name} · features: ${data.features}`);
      return true;
    }
    _state = { ..._state, valid: false, checked: true, error: data.error || 'Licence invalid' };
    console.error(`[{{ product.code }}] ✗ ${_state.error}`);
    return false;
  } catch (e) {
    _state = { ..._state, valid: false, checked: true, error: e.message };
    return false;
  }
}

function hasFeature(code) { return _state.features.includes(code); }
function getState()       { return { ..._state }; }
function isValid()        { return _state.valid; }

async function initLicense() {
  const ok = await validateLicense();
  setInterval(async () => {
    try {
      await axios.post(`${LICENSE_SERVER}/api/heartbeat`,
        { license_key: _state.licenseKey, activation_id: _state.activationId },
        { timeout: VALIDATE_TIMEOUT });
    } catch {}
  }, HEARTBEAT_MS);
  return ok;
}

module.exports = { initLicense, isValid, hasFeature, getState };

# Add this route to your Flask app (e.g. in app.py or license_check.py)
# It receives the key from the activation page, saves it, and re-validates.

from flask import request, jsonify

@app.route('/licence-activate', methods=['POST'])
def licence_activate():
    """Web endpoint — lets users enter their licence key via the browser."""
    data = request.get_json() or {}
    key  = data.get('license_key', '').strip().upper()

    if not key or len(key) < 10:
        return jsonify({'success': False, 'error': 'Invalid key format'}), 400

    # Save to license.txt
    try:
        with open(LICENSE_FILE, 'w') as f:
            f.write(f"# {{ product.name }} Licence Key\n{key}\n")
    except Exception as e:
        return jsonify({'success': False, 'error': f'Could not save key: {e}'}), 500

    # Re-validate immediately
    ok = validate_license(key=key)
    if ok:
        return jsonify({
            'success': True,
            'message': f'Licence activated for {get_state().get("customer", "your account")}',
            'tier': get_state().get('tier'),
            'features': get_state().get('features', []),
        }), 200

    return jsonify({
        'success': False,
        'error': get_state().get('error', 'Licence validation failed'),
    }), 200

How it works

1. User visits your app with no licence → sees license_invalid.html

2. They paste their key and click Activate

3. Page calls POST /licence-activate on your app

4. Your app saves the key to license.txt and calls the licence server

5. If valid → user is redirected to / and the app unlocks immediately

6. If invalid → error is shown inline, they can try again

Feature Gating Example

Use has_feature() anywhere in your app to show/hide functionality based on what the licence includes:

from license_check import has_feature, get_state

# Guard a route
@app.route('/dispatch')
def dispatch():
    if not has_feature('dispatch'):
        return "Feature not included in your licence", 403
    return render_template('dispatch.html')

# Pass features to a template
@app.route('/dashboard')
def dashboard():
    state = get_state()
    return render_template('dashboard.html',
        features=state['features'],
        customer=state['customer'],
        tier=state['tier'],
    )
